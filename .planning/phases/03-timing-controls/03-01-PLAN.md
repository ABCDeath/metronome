    Create vitest.config.ts at project root. Import defineConfig from 'vitest/config' and svelte from '@sveltejs/vite-plugin-svelte'. Export defineConfig with plugins: [svelte()], test.include: ['src/**/*.test.ts'], test.environment: 'node'.

    Create src/lib/pattern.ts with these named exports (per D-05, D-07, D-08):
    - type Subdivision = 'quarter' | 'eighth' | 'triplet' | 'sixteenth'
    - type BeatPosition = { voice: number }
    - type Track = { stepCount: number; subdivision: Subdivision; denominator: number; beats: BeatPosition[] }
    - type PatternState = { bpm: number; tracks: Track[]; accentFreqHz: number; accentAmpMillis: number }
    - const SUBDIV_MULT: Record<Subdivision, number> = { quarter: 1, eighth: 2, triplet: 3, sixteenth: 4 }
    - function computeStepInterval(bpm, denominator, subdivMult): returns (60.0 / bpm) * (4 / denominator) / subdivMult
    - function rebuildBeats(stepCount): returns Array.from({ length: stepCount }, (_, i) => ({ voice: i === 0 ? 1 : 0 }))
    - function defaultPatternState(): returns { bpm: 120, tracks: [{ stepCount: 4, subdivision: 'quarter', denominator: 4, beats: [{voice:1},{voice:0},{voice:0},{voice:0}] }], accentFreqHz: 1400, accentAmpMillis: 1300 }

    accentFreqHz and accentAmpMillis are top-level PatternState fields (not nested in Track) because accent settings are global, not per-track. accentAmpMillis = amplitude multiplier times 1000 (integer for Atomics.store).
  </action>
  <acceptance_criteria>
